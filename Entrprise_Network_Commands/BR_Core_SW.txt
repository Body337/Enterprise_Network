enable
configure terminal

hostname BR-Core-SW

! 1. Create the Branch Data and Management VLANs
vlan 30
 name BR-Data
vlan 99
 name BR-Mgmt
exit

! 2. Set Spanning Tree to Rapid mode (matches HQ)
spanning-tree mode rapid-pvst

! 3. Configure the Trunk port going up to the BR-FortiGate (Ethernet 0/0)
interface Ethernet 0/0
 description LINK_TO_BR_FGT_PORT3
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 30,99
 switchport trunk native vlan 99
 no shutdown
exit

! 4. Configure the Access port for the Branch PC (BR-PC1 on Ethernet 0/1)
interface Ethernet 0/1
 description LINK_TO_BR_PC1
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
 no shutdown
exit
