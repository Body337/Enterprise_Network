enable
configure terminal

hostname HQ-Core-SW

! 1. Create the Enterprise VLANs
vlan 10
 name HQ-Data
vlan 20
 name HQ-Servers
vlan 99
 name HQ-Mgmt
exit

! 2. Configure Spanning Tree Protocol (Rapid-PVST+ for faster convergence)
spanning-tree mode rapid-pvst
spanning-tree vlan 10,20,99 root primary

! 3. Configure the Trunk port going up to the HQ-FortiGate (Port 3)
interface Ethernet 0/0
 description LINK_TO_HQ_FGT_PORT3
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,99
 switchport trunk native vlan 99
 no shutdown
exit

! 4. Configure the Trunk port going down to the Access Switch
interface Ethernet 0/1
 description LINK_TO_HQ_ACCESS_SW
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,99
 switchport trunk native vlan 99
 no shutdown
exit

! 5. Configure the Access port for the FortiManager (VLAN 20 Server Space)
interface Ethernet 0/3
 description LINK_TO_FORTIMANAGER
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
 no shutdown
exit
